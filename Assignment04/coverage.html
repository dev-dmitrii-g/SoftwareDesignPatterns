package equipmentreserver

import (
        "database/sql"
        "errors"
        "fmt"
)

const (
        New      = "NEW"
        Reserved = "RESERVED"
        Released = "RELEASED"
)

type Reservation struct{ AdmissionID, Item, Status string }

func (v *Reservation) Transition(to string) error {
        if (v.Status == New && to == Reserved) || (v.Status == Reserved && to == Released) {
                v.Status = to
                return nil
        }
        return fmt.Errorf("equipment: illegal transition %s -> %s", v.Status, to)
}

type Rule struct{}

func (Rule) Check(item string) error {
        if item == "" {
                return errors.New("equipment name is required")
        }
        return nil
}

type Repository struct{ DB *sql.DB }

func (r Repository) Get(id string) (Reservation, error) {
        var v Reservation
        err := r.DB.QueryRow(`SELECT admission_id,item,status FROM equipment WHERE admission_id=?`, id).Scan(&v.AdmissionID, &v.Item, &v.Status)
        return v, err
}
func (r Repository) SetStock(item string, n int) error {
        if n < 0 {
                return errors.New("stock cannot be negative")
        }
        _, err := r.DB.Exec(`INSERT INTO stock(kind,name,quantity) VALUES('equipment',?,?) ON CONFLICT(kind,name) DO UPDATE SET quantity=excluded.quantity`, item, n)
        return err
}
func (r Repository) Reserve(v Reservation) error {
        tx, err := r.DB.Begin()
        if err != nil {
                return err
        }
        defer tx.Rollback()
        res, err := tx.Exec(`UPDATE stock SET quantity=quantity-1 WHERE kind='equipment' AND name=? AND quantity>0`, v.Item)
        if err != nil {
                return err
        }
        n, err := res.RowsAffected()
        if err != nil {
                return err
        }
        if n == 0 {
                return fmt.Errorf("equipment %q unavailable", v.Item)
        }
        if _, err = tx.Exec(`INSERT INTO equipment(admission_id,item,status) VALUES(?,?,?)`, v.AdmissionID, v.Item, v.Status); err != nil {
                return err
        }
        return tx.Commit()
}
func (r Repository) Release(v Reservation) error {
        tx, err := r.DB.Begin()
        if err != nil {
                return err
        }
        defer tx.Rollback()
        res, err := tx.Exec(`UPDATE equipment SET status=? WHERE admission_id=? AND status=?`, Released, v.AdmissionID, Reserved)
        if err != nil {
                return err
        }
        n, err := res.RowsAffected()
        if err != nil {
                return err
        }
        if n != 1 {
                return errors.New("reservation is not active")
        }
        if _, err = tx.Exec(`UPDATE stock SET quantity=quantity+1 WHERE kind='equipment' AND name=?`, v.Item); err != nil {
                return err
        }
        return tx.Commit()
}

type Reserver struct {
        Rule       Rule
        Repository Repository
}

func (s Reserver) Reserve(id, item string) error {
        if err := s.Rule.Check(item); err != nil {
                return err
        }
        v := Reservation{id, item, New}
        if err := v.Transition(Reserved); err != nil {
                return err
        }
        return s.Repository.Reserve(v)
}
func (s Reserver) Release(id string) error {
        v, err := s.Repository.Get(id)
        if err != nil {
                return err
        }
        if err = v.Transition(Released); err != nil {
                return err
        }
        return s.Repository.Release(v)
}

type View struct {
        Admission                      Admission
        Insurance, Equipment, Pharmacy string
}

type AdminFacade struct{ Intake IntakeFacade }

func (f AdminFacade) Inspect(id string) (View, error) {
        a, err := f.Intake.Gateway.Get(id)
        if err != nil {
                return View{}, err
        }
        v := View{Admission: a, Insurance: "NONE", Equipment: "NONE", Pharmacy: "NONE"}
        i, e := f.Intake.Insurance.Repository.Get(id)
        if e != nil && e != sql.ErrNoRows {
                return View{}, e
        }
        if e == nil {
                v.Insurance = i.Status
        }
        b, e := f.Intake.Equipment.Repository.Get(id)
        if e != nil && e != sql.ErrNoRows {
                return View{}, e
        }
        if e == nil {
                v.Equipment = b.Status
        }
        p, e := f.Intake.Pharmacy.Repository.Get(id)
        if e != nil && e != sql.ErrNoRows {
                return View{}, e
        }
        if e == nil {
                v.Pharmacy = p.Status
        }
        return v, nil
}
func (f AdminFacade) Compensate(id string) error {
        v, err := f.Inspect(id)
        if err != nil {
                return err
        }
        if v.Admission.Status == Completed {
                return errors.New("completed admission cannot be manually compensated")
        }
        if v.Pharmacy == pharmacydispatcher.Dispatched {
                return errors.New("dispatched medicine cannot be manually compensated")
        }
        if v.Admission.Status == Compensated {
                return nil
        }
        return f.Intake.compensate(id, "manual compensation")
}

import "database/sql"

const (
        PendingA     = "PENDING_A"
        PendingB     = "PENDING_B"
        PendingC     = "PENDING_C"
        Completed    = "COMPLETED"
        Compensating = "COMPENSATING"
        Compensated  = "COMPENSATED"
)

type Admission struct{ ID, IIN, Coverage, Equipment, Medicine, Status, Reason string }

type Gateway struct{ DB *sql.DB }

func (g Gateway) Create(a Admission) error {
        _, err := g.DB.Exec(`INSERT INTO admissions(id,iin,coverage,equipment,medicine,status,reason) VALUES(?,?,?,?,?,?,?)`, a.ID, a.IIN, a.Coverage, a.Equipment, a.Medicine, PendingA, "")
        return err
}
func (g Gateway) Get(id string) (Admission, error) {
        var a Admission
        err := g.DB.QueryRow(`SELECT id,iin,coverage,equipment,medicine,status,reason FROM admissions WHERE id=?`, id).Scan(&a.ID, &a.IIN, &a.Coverage, &a.Equipment, &a.Medicine, &a.Status, &a.Reason)
        return a, err
}
func (g Gateway) List() ([]Admission, error) {
        rows, err := g.DB.Query(`SELECT id,iin,coverage,equipment,medicine,status,reason FROM admissions ORDER BY rowid DESC`)
        if err != nil {
                return nil, err
        }
        defer rows.Close()
        items := []Admission{}
        for rows.Next() {
                var a Admission
                if err := rows.Scan(&a.ID, &a.IIN, &a.Coverage, &a.Equipment, &a.Medicine, &a.Status, &a.Reason); err != nil {
                        return nil, err
                }
                items = append(items, a)
        }
        return items, rows.Err()
}
func (g Gateway) Update(id, status, reason string) error {
        _, err := g.DB.Exec(`UPDATE admissions SET status=?,reason=? WHERE id=?`, status, reason, id)
        return err
}

package facades

import (
        "database/sql"
        "errors"
        "fmt"

        "hospitaladmission/internal/equipmentreserver"
        "hospitaladmission/internal/insuranceverifier"
        "hospitaladmission/internal/pharmacydispatcher"
)

type IntakeFacade struct {
        Gateway   Gateway
        Insurance insuranceverifier.Verifier
        Equipment equipmentreserver.Reserver
        Pharmacy  pharmacydispatcher.Dispatcher
}

func (f IntakeFacade) Start(a Admission) error {
        if a.ID == "" {
                return errors.New("admission ID is required")
        }
        if err := f.Insurance.Rule.Check(a.IIN, a.Coverage); err != nil {
                return err
        }
        if err := f.Equipment.Rule.Check(a.Equipment); err != nil {
                return err
        }
        if err := f.Pharmacy.Rule.Check(a.Medicine); err != nil {
                return err
        }
        if err := f.Gateway.Create(a); err != nil {
                return fmt.Errorf("create admission: %w", err)
        }
        return f.ensureA(a)
}
func (f IntakeFacade) Admit(a Admission) error {
        if err := f.Start(a); err != nil {
                return err
        }
        return f.Resume(a.ID)
}
func (f IntakeFacade) ensureA(a Admission) error {
        v, err := f.Insurance.Repository.Get(a.ID)
        if err == sql.ErrNoRows {
                if err = f.Insurance.Verify(a.ID, a.IIN, a.Coverage); err != nil {
                        return err
                }
        } else if err != nil {
                return err
        } else if v.Status != insuranceverifier.Verified {
                return fmt.Errorf("insurance state %s cannot continue", v.Status)
        }
        return f.Gateway.Update(a.ID, PendingB, "")
}
func (f IntakeFacade) Resume(id string) error {
        a, err := f.Gateway.Get(id)
        if err != nil {
                return err
        }
        switch a.Status {
        case Completed:
                return nil
        case Compensated:
                return errors.New("admission was compensated")
        case Compensating:
                return f.compensate(a.ID, a.Reason)
        case PendingA, PendingB, PendingC:
        default:
                return fmt.Errorf("unknown admission status %s", a.Status)
        }
        if a.Status == PendingA {
                if err = f.ensureA(a); err != nil {
                        return err
                }
        }
        v, err := f.Equipment.Repository.Get(id)
        if err == sql.ErrNoRows {
                if err = f.Equipment.Reserve(id, a.Equipment); err != nil {
                        return f.fail(id, err)
                }
        } else if err != nil {
                return err
        } else if v.Status != equipmentreserver.Reserved {
                return fmt.Errorf("equipment state %s cannot continue", v.Status)
        }
        if err = f.Gateway.Update(id, PendingC, ""); err != nil {
                return err
        }
        p, err := f.Pharmacy.Repository.Get(id)
        if err == sql.ErrNoRows {
                if err = f.Pharmacy.Dispatch(id, a.Medicine); err != nil {
                        return f.fail(id, err)
                }
        } else if err != nil {
                return err
        } else if p.Status != pharmacydispatcher.Dispatched {
                return fmt.Errorf("pharmacy state %s cannot continue", p.Status)
        }
        return f.Gateway.Update(id, Completed, "")
}
func (f IntakeFacade) fail(id string, cause error) error {
        if err := f.compensate(id, cause.Error()); err != nil {
                return fmt.Errorf("%v; compensation failed: %w", cause, err)
        }
        return cause
}
func (f IntakeFacade) compensate(id, reason string) error {
        if err := f.Gateway.Update(id, Compensating, reason); err != nil {
                return err
        }
        b, err := f.Equipment.Repository.Get(id)
        if err != nil && err != sql.ErrNoRows {
                return err
        }
        if err == nil && b.Status == equipmentreserver.Reserved {
                if err = f.Equipment.Release(id); err != nil {
                        return err
                }
        }
        a, err := f.Insurance.Repository.Get(id)
        if err != nil && err != sql.ErrNoRows {
                return err
        }
        if err == nil && a.Status == insuranceverifier.Verified {
                if err = f.Insurance.Reverse(id); err != nil {
                        return err
                }
        }
        return f.Gateway.Update(id, Compensated, reason)
}

package insuranceverifier

import (
        "database/sql"
        "errors"
        "fmt"
)

const (
        New        = "NEW"
        Verified   = "VERIFIED"
        Reversed   = "REVERSED"
        NoCoverage = "NONE"
        GOBMP      = "GOBMP"
        OSMS       = "OSMS"
        DMS        = "DMS"
)

type Verification struct{ AdmissionID, IIN, Coverage, Status string }

func (v *Verification) Transition(to string) error {
        if (v.Status == New && to == Verified) || (v.Status == Verified && to == Reversed) {
                v.Status = to
                return nil
        }
        return fmt.Errorf("insurance: illegal transition %s -> %s", v.Status, to)
}

type Rule struct{}

func (Rule) Check(iin, coverage string) error {
        if len(iin) != 12 {
                return errors.New("IIN must contain 12 digits")
        }
        for _, r := range iin {
                if r < '0' || r > '9' {
                        return errors.New("IIN must contain 12 digits")
                }
        }
        switch coverage {
        case GOBMP, OSMS, DMS:
                return nil
        case NoCoverage:
                return errors.New("no coverage selected")
        default:
                return errors.New("choose GOBMP, OSMS, or DMS coverage")
        }
}

type Repository struct{ DB *sql.DB }

func (r Repository) Save(v Verification) error {
        _, err := r.DB.Exec(`INSERT INTO insurance(admission_id,iin,coverage,status) VALUES(?,?,?,?)
        ON CONFLICT(admission_id) DO UPDATE SET status=excluded.status`, v.AdmissionID, v.IIN, v.Coverage, v.Status)
        return err
}
func (r Repository) Get(id string) (Verification, error) {
        var v Verification
        err := r.DB.QueryRow(`SELECT admission_id,iin,coverage,status FROM insurance WHERE admission_id=?`, id).Scan(&v.AdmissionID, &v.IIN, &v.Coverage, &v.Status)
        return v, err
}

type Verifier struct {
        Rule       Rule
        Repository Repository
}

func (s Verifier) Verify(id, iin, coverage string) error {
        if err := s.Rule.Check(iin, coverage); err != nil {
                return err
        }
        v := Verification{id, iin, coverage, New}
        if err := v.Transition(Verified); err != nil {
                return err
        }
        return s.Repository.Save(v)
}
func (s Verifier) Reverse(id string) error {
        v, err := s.Repository.Get(id)
        if err != nil {
                return err
        }
        if err = v.Transition(Reversed); err != nil {
                return err
        }
        return s.Repository.Save(v)
}

package store

import (
        "database/sql"
        "fmt"

        _ "github.com/mattn/go-sqlite3"
)

func Open(path string) (*sql.DB, error) {
        db, err := sql.Open("sqlite3", path)
        if err != nil {
                return nil, err
        }
        db.SetMaxOpenConns(1)
        if _, err = db.Exec(`PRAGMA busy_timeout = 5000;
        CREATE TABLE IF NOT EXISTS insurance (admission_id TEXT PRIMARY KEY, iin TEXT NOT NULL, coverage TEXT NOT NULL, status TEXT NOT NULL);
        CREATE TABLE IF NOT EXISTS equipment (admission_id TEXT PRIMARY KEY, item TEXT NOT NULL, status TEXT NOT NULL);
        CREATE TABLE IF NOT EXISTS pharmacy (admission_id TEXT PRIMARY KEY, medicine TEXT NOT NULL, status TEXT NOT NULL);
        CREATE TABLE IF NOT EXISTS stock (kind TEXT NOT NULL, name TEXT NOT NULL, quantity INTEGER NOT NULL CHECK(quantity >= 0), PRIMARY KEY(kind, name));
        CREATE TABLE IF NOT EXISTS admissions (id TEXT PRIMARY KEY, iin TEXT NOT NULL, coverage TEXT NOT NULL, equipment TEXT NOT NULL, medicine TEXT NOT NULL, status TEXT NOT NULL, reason TEXT NOT NULL DEFAULT '');
        INSERT OR IGNORE INTO stock(kind,name,quantity) VALUES('equipment','wheelchair',5);
        INSERT OR IGNORE INTO stock(kind,name,quantity) VALUES('pharmacy','paracetamol',5);`); err != nil {
                db.Close()
                return nil, fmt.Errorf("initialize database: %w", err)
        }
        for _, table := range []string{"insurance", "admissions"} {
                if err := migrateCoverage(db, table); err != nil {
                        db.Close()
                        return nil, fmt.Errorf("migrate %s: %w", table, err)
                }
        }
        return db, nil
}

func migrateCoverage(db *sql.DB, table string) error {
        rows, err := db.Query("PRAGMA table_info(" + table + ")")
        if err != nil {
                return err
        }
        hasPolicy, hasCoverage := false, false
        for rows.Next() {
                var position, required, primaryKey int
                var name, dataType string
                var defaultValue sql.NullString
                if err := rows.Scan(&position, &name, &dataType, &required, &defaultValue, &primaryKey); err != nil {
                        rows.Close()
                        return err
                }
                if name == "policy" {
                        hasPolicy = true
                }
                if name == "coverage" {
                        hasCoverage = true
                }
        }
        if err := rows.Err(); err != nil {
                rows.Close()
                return err
        }
        if err := rows.Close(); err != nil {
                return err
        }
        if hasPolicy && !hasCoverage {
                _, err = db.Exec("ALTER TABLE " + table + " RENAME COLUMN policy TO coverage")
                return err
        }
        return nil
}
